# [28. Find the Index of the First Occurrence in a String](https://leetcode.com/problems/find-the-index-of-the-first-occurrence-in-a-string)






## Brute-Force

```cpp
class Solution {
public:
    int strStr(string haystack, string needle) {
        int tar_len=needle.length();
        int len=haystack.length();
        
        int j=0;
        for(int i=0; i<len; i++){
            if(haystack[i]==needle[j]){
                while(j<tar_len){
                    if(haystack[i+j] == needle[j]){
                        j++;
                    }else{
                        j=0;
                        break;
                    }
                }
                
                if(j!=0){
                    return i;
                }
            }
        }
        
        return -1;
    }
};
```

## Fixed-sized Slide window 

```cpp
class Solution {
public:
    int strStr(string haystack, string needle) {
        int len = haystack.length();
        int tar_len = needle.length();

        for (int i = 0; i <= len - tar_len; i++) {

            int j = 0;

            while (j < tar_len) {
                if (haystack[i + j] != needle[j])
                    break;

                j++;
            }

            if (j == tar_len)
                return i;
        }

        return -1;
    }
};

```

## Rabin-Karp String Matching Algorithm

```cpp
class Solution {
public:
    int strStr(string haystack, string needle) {
        int n = haystack.size(), m = needle.size();
        const int mod = (1 << 31) - 1;
        long long target = 0, sha = 0, multi = 1;
        for (int i = 0; i < m; ++i) {
            target = (target * 256 + needle[i]) % mod;
        }
        for (int i = 1; i < m; ++i) {
            multi = multi * 256 % mod;
        }
        int left = 0;
        for (int right = 0; right < n; ++right) {
            sha = (sha * 256 + haystack[right]) % mod;
            if (right - left + 1 < m) {
                continue;
            }
            if (sha == target && haystack.substr(left, m) == needle) {
                return left;
            }
            sha = (sha - haystack[left] * multi % mod + mod) % mod;
            ++left;
        }
        return -1;
    }
};
```

## KMP (Knuth-Morris-Pratt) 

```cpp
class Solution {
public:
    int strStr(string haystack, string needle) {
        int n = haystack.size();
        int m = needle.size();

        vector<int> lps(m, 0);

        // Build LPS
        for (int i = 1, j = 0; i < m; ) {
            if (needle[i] == needle[j]) {
                lps[i++] = ++j;
            }
            else if (j > 0) {
                j = lps[j - 1];
            }
            else {
                i++;
            }
        }

        // Search
        for (int i = 0, j = 0; i < n; ) {
            if (haystack[i] == needle[j]) {
                i++;
                j++;

                if (j == m)
                    return i - m;
            }
            else if (j > 0) {
                j = lps[j - 1];
            }
            else {
                i++;
            }
        }

        return -1;
    }
};
```

## STL library

```cpp
class Solution {
public:
    int strStr(std::string haystack, std::string needle) {
        for (int i = 0; i <= haystack.length() - needle.length(); ++i) {
            if (haystack.substr(i, needle.length()) == needle) {
                return i;
            }
        }
        return -1;
    }
};
```

## 📊 Complexity Comparison

| Approach | Main Idea | Time Complexity | Space Complexity | Notes |
|---|---|---:|---:|---|
| **Brute Force** | Compare `needle` from every possible index | `O(n × m)` | `O(1)` | Simple, direct character comparison |
| **Fixed-Size Sliding Window** | Consider every window of size `m` | `O(n × m)` | `O(1)` | Window size = `needle.length()` |
| **KMP** | Use LPS to avoid repeated comparisons | `O(n + m)` | `O(m)` | Best worst-case complexity |
| **STL `substr()`** | Create substring and compare with `needle` | `O(n × m)`* | `O(m)`* | Very simple, library handles comparison |

### Notation

- `n` = length of `haystack`
- `m` = length of `needle`
